\begin{abstract}
We study the classical multi-item monopoly problem for a single additive buyer with $m \ge 1$ independent nonnegative item valuations $\bm{X} = (X_1,\dots,X_m)$, bounding the worst-case approximation ratio of the better of selling separately ($\SREV(\bm{X})$) and grand bundling ($\BREV(\bm{X})$) against the optimal truthful mechanism revenue ($\REV(\bm{X})$). Giannakopoulos and Hahn (\texttt{arXiv:2609.35711v3}, Nov.\ 2026) established the interval $[2.19027, 2.50]$ (specifically upper bound $\frac{5}{2} = 2.50$ in Proposition~4.6, with Appendix~C.4.1 case values implying $\frac{1499}{600} \approx 2.49833$, and deterministic lower bound $\frac{2026}{925} \approx 2.190270 > 2.19$ in Appendix~D, whose published $3{,}000$-item menu has certified limiting ratio in $[2.190949496843, 2.190949496844]$), while noting that sharper unpublished bounds exist on both sides. We tighten both endpoints simultaneously to the certified interval
\[
  \frac{439}{200} = 2.195 \;<\; \sup_{m,\bm{X}} \frac{\DREV(\bm{X})}{\max\{\SREV(\bm{X}), \BREV(\bm{X})\}}
  \;\le\; \sup_{m,\bm{X}} \frac{\REV(\bm{X})}{\max\{\SREV(\bm{X}), \BREV(\bm{X})\}}
  \;\le\; \frac{62}{25} = 2.48.
\]
In \textbf{Part~I} (Upper Bound $\frac{62}{25} = 2.48$), building on the capped tail-integral and Laplace-transform inequalities of Giannakopoulos and Hahn, we replace static item-separating weights with pointwise time-varying Jensen weights $\pi_j(t) \propto \frac{r_j}{c_j+t}$ (equivalently, a direct chord bound on $\phi(\sum_j \bar{F}_j(c_j+t))$), collapsing all item tails onto the Stieltjes transform $U_1(t) = \int (c+t)^{-1}\,d\mu(c)$ of the cap probability measure $\mu$. Using concavity of $F(t) = 1/U_1(t)$ (a classical property of reciprocals of Stieltjes transforms / complete Bernstein functions, proved directly via Cauchy--Schwarz) yields a three-moment harmonic-secant envelope at $(l_1,l_2,l_3) = (\frac{3}{50}, \frac{7}{20}, \frac{9}{5})$ certified over the full 3D moment lens by an exact rational binary-tree cover ($\LeafCount$ leaves, $\DisjointCount$ disjoint boxes) and dual-verified by standalone Rust IEEE-754 interval ($\le \RustCertUB$) and Python \texttt{mpmath.iv} $35$-digit ($\le \PyCertUB$) verifiers.

In \textbf{Part~II} (Lower Bound $\frac{439}{200} = 2.195$), we optimize within the entry-fee plus concentrating-bundle template of Giannakopoulos and Hahn by eliminating the free item group ($p_E = 0$) and distributing separate revenue across an exact-rational \emph{all-paid} eight-group equal-revenue spectrum ($N_a = 5{,}000$ items per group on the $1/100$ grid, entry fee $F = 7$, $S_A = \frac{1949}{400} = 4.8725$). Dual outward-rounded Rust and Python verifiers certify the exact degree-$699$ surplus polynomials, all $5{,}013$ finite-grid grand-bundle checks covering $(S_A, 55]$, and a Chernoff MGF tail covering $[55,\infty)$, achieving limiting ratio $\ge \RustCertLB$ and an explicit finite deterministic DSIC/IR witness at $N_b = 2^{46800}$, $\delta = S_A / 10^8$ strictly exceeding $\frac{439}{200}$ under every buyer tie-breaking rule.

Both parts are accompanied by a Lean~4 + Mathlib formalization (\texttt{MathForgeMechDesign}, $55$ exported theorems, zero \texttt{sorry}, zero \texttt{native\_decide}) verifying reciprocal concavity, the discrete cap measure, the 3D moment polytope and Farkas decoupling, exact-rational binary tree coverage and $32$-box disjointness, Time-Varying Jensen Alignment (with concrete $m=1$ non-vacuity witness), 8-group PMF/$\REV$/$\SREV$, bundle-$B$ variance and Chebyshev bounds, the explicit $N_b = 2^{46800}$ witness, and DSIC/IR tie-breaking invariance.
\end{abstract}

\section{Introduction and Summary of Results}

Let $X_1,\dots,X_m$ be independent nonnegative random variables representing a single additive buyer's valuations for $m \ge 1$ items. Denote by $\REV(\bm{X})$ the supremum expected revenue achievable by any incentive-compatible (Bayesian or dominant-strategy, equivalent for a single buyer) and individually rational (possibly randomized) mechanism, by $\DREV(\bm{X}) \le \REV(\bm{X})$ the supremum over deterministic truthful mechanisms, by $\SREV(\bm{X}) = \sum_{j=1}^m \REV(X_j)$ the optimal revenue from selling each item separately, and by $\BREV(\bm{X}) = \sup_{p \ge 0} p\Pr[\sum_{j=1}^m X_j \ge p]$ the optimal revenue from selling the grand bundle.

Starting from the $6$-approximation of Babaioff, Immorlica, Lucier, and Weinberg~\cite{BILW20}, the upper bound $C$ in $\REV(\bm{X}) \le C\max\{\SREV(\bm{X}),\BREV(\bm{X})\}$ progressed through $5.2$~\cite{MaSimchiLevi21}, $3.52$~\cite{CaiEtAl26}, $2.645$~\cite{Huang26}, and $\frac{5}{2} = 2.50$~\cite[Proposition~4.6]{GiannakopoulosHahn26}. On the lower-bound side, Rubinstein~\cite{Rubinstein16} established $2$, Huang~\cite{Huang26} proved $2.11$ via a deterministic three-bundle partition mechanism, and Giannakopoulos and Hahn~\cite[Theorem~D.1, Appendix~D]{GiannakopoulosHahn26} established a deterministic lower bound exceeding $2.19$ ($\frac{2026}{925} \approx 2.190270$ displayed). Table~\ref{tab:md-baselines} distinguishes published stated bounds from our certified reevaluations and controls (audited on November~25, 2026).

\begin{theorem}[Main Upper Bound]\label{thm:ub-main}
For every $m \ge 1$ and any independent nonnegative random variables $X_1,\dots,X_m$,
\[
  \REV(\bm{X}) \le \BoundFrac \cdot \max\{\SREV(\bm{X}), \BREV(\bm{X})\} = \BoundConst \cdot \max\{\SREV(\bm{X}), \BREV(\bm{X})\}.
\]
\end{theorem}

\begin{theorem}[Main Deterministic Lower Bound]\label{thm:lb-main}
There exists a finite product distribution $\bm{X} = (X_1,\dots,X_m)$ of independent nonnegative random variables (explicitly realized with $40{,}000 + N_b^2$ items at $N_b = 2^{46800}$) and a deterministic, dominant-strategy incentive-compatible (DSIC) and individually rational (IR) mechanism $\mathcal{M}$ whose expected revenue satisfies
\[
  \REV(\bm{X}) \ge \DREV(\bm{X}) \ge \REV(\mathcal{M};\bm{X}) > \TargetFrac \max\{\SREV(\bm{X}), \BREV(\bm{X})\} = \TargetDec \max\{\SREV(\bm{X}), \BREV(\bm{X})\}
\]
under every buyer tie-breaking rule.
\end{theorem}

\section{Preliminaries, Domain Reduction, and Normalization}\label{sec:ub-prelim}

We take as our starting point Lemmas~4.4 and~4.5 of Giannakopoulos and Hahn~\cite{GiannakopoulosHahn26} (which are proved on the finite discrete unit $K$-grid $I_K := \{0, 1/K, \dots, 1\}$). For each item $j \in [m]$, let $\bar{F}_j(u) := \Pr[X_j \ge u]$ (using $\Pr[X_j > u]$ alters $\bar{F}_j$ on at most a countable set and leaves both the monopoly supremum and all Lebesgue integrals unchanged), and let $r_j := \REV(X_j) = \sup_{u > 0} u\bar{F}_j(u)$. Write $S := \SREV(\bm{X}) = \sum_{j=1}^m r_j$, $B := \BREV(\bm{X})$, and $M := \max\{S, B\}$.

\paragraph{Boundary cases, reduction from $[0,\infty)^m$ to $I_K^m$, and normalization.}
\begin{itemize}
  \item If $M = \infty$ (note $B \le mS$ by $\Pr[\sum_j X_j \ge p] \le \sum_j \Pr[X_j \ge p/m]$, so $M = \infty \iff S = \infty$), Theorem~\ref{thm:ub-main} holds trivially in $[0,\infty]$.
  \item If $M = 0$, then $r_j = 0$ for every $j$, so $X_j = 0$ almost surely and $\REV(\bm{X}) = 0$.
  \item Assume $0 < M(\bm{X}) < \infty$. For any $H > 0$, let $X_j^{(H)} := X_j \wedge H \in [0,H]$; clearly $M(\bm{X}^{(H)}) \le M(\bm{X})$. Given any IC/IR mechanism $(a,p)$ on $[0,\infty)^m$, set $\tilde{p}(\bm{x}) := p(\bm{x}) - p(\bm{0})$. IR at $\bm{0}$ gives $p(\bm{0}) \le 0$ (so $\tilde{p} \ge p$), IC of type $\bm{0}$ reporting $\bm{x}$ gives $\tilde{p}(\bm{x}) \ge 0$, constant payment shifts preserve IC, and IC of type $\bm{x}$ reporting $\bm{0}$ gives $\bm{x}\cdot a(\bm{x}) - \tilde{p}(\bm{x}) \ge \bm{x}\cdot a(\bm{0}) \ge 0$, preserving IR (kernel-proved in \texttt{shifted\_payment\_ic\_ir}). Restricting $(a,\tilde{p})$ to $[0,H]^m$ yields $\REV(\bm{X}^{(H)}) \ge \E[\tilde{p}(\bm{X})\mathbf{1}_{\{\max_j X_j \le H\}}]$, so monotone convergence (requiring no finiteness of $\E[X_j]$) gives $\REV(\bm{X}) \le \liminf_{H\to\infty}\REV(\bm{X}^{(H)})$. Rescaling $\bm{Y} := \bm{X}^{(H)}/H \in [0,1]^m$ by positive homogeneity and setting $\bm{Y}^{(K)} := \lfloor K\bm{Y}\rfloor/K \in I_K^m$, Proposition~2.2 (eqs.~(6)--(7)) of~\cite{GiannakopoulosHahn26} yields $M(\bm{Y}^{(K)}) \le M(\bm{Y})$ and $\REV(\bm{Y}) \le \REV(\bm{Y}^{(K)}) + 2m/\sqrt{K}$. Letting $K \to \infty$ then $H \to \infty$ reduces Theorem~\ref{thm:ub-main} to $I_K^m$.
  \item Dropping items with $r_j = 0$ ($0$ a.s.) leaves $S > 0$ and $B \ge \max_j r_j > 0$. Rescaling $X_j := Y_j^{(K)} / M(\bm{Y}^{(K)})$ yields without loss of generality a finite-grid instance satisfying
  \begin{equation}\label{eq:norm}
    M = \max\{S, B\} = 1, \qquad 0 < S \le 1, \qquad 0 < B \le 1, \qquad 0 < r_j \le 1 \text{ for all } j \in [m].
  \end{equation}
\end{itemize}

\begin{lemma}[Existence of Finite Positive Caps]\label{lem:finite-caps}
Under~\eqref{eq:norm}, for every $j \in [m]$, the minimal cap $c_j := \inf\{c \ge 0 : \int_0^c \bar{F}_j(u)\,du = r_j\}$ satisfies $0 < r_j \le c_j < \infty$ and $\E[X_j \wedge c_j] = \int_0^{c_j} \bar{F}_j(u)\,du = r_j$.
\end{lemma}

\begin{proof}
The function $H_j(c) := \int_0^c \bar{F}_j(u)\,du$ is $1$-Lipschitz, nondecreasing, vanishes at $0$, satisfies $H_j(c) \le c$, and increases to $\E[X_j] \ge r_j$. If $\E[X_j] > r_j$, the intermediate value theorem gives $r_j \le c_j < \infty$. If $\E[X_j] = r_j \in (0,\infty)$, pick $u_n > 0$ with $u_n\bar{F}_j(u_n) \to r_j$; neither $u_{n_k} \to \infty$ (since $u\bar{F}_j(u) \le \E[X_j\mathbf{1}\{X_j \ge u\}] \to 0$) nor $u_{n_k} \to 0$ (since $u\bar{F}_j(u) \le u \to 0$) is possible, so a subsequence converges to $u^* \in (0,\infty)$. Upper semicontinuity gives $r_j \le u^*\Pr[X_j \ge u^*] \le \E[X_j\mathbf{1}\{X_j \ge u^*\}] \le \E[X_j] = r_j$, forcing $X_j \in \{0, u^*\}$ a.s.\ and $H_j(u^*) = r_j$, hence $c_j \le u^* < \infty$.
\end{proof}

Set $\phi(x) := x - 1 + e^{-x}$ ($x \ge 0$) and $L(y) := -\ln(\int_1^\infty e^{-yx} x^{-2}\,dx) = y - \ln(\int_0^\infty e^{-yt}(1+t)^{-2}\,dt)$ ($y > 0$). For finite caps $c_j > 0$ with $\E[X_j \wedge c_j] = r_j$, Giannakopoulos and Hahn~\cite[Lemmas~4.4 and~4.5]{GiannakopoulosHahn26} establish:
\begin{align}
  \REV(\bm{X}) &\le S + \int_0^\infty \phi\!\left(\sum_{j=1}^m \bar{F}_j(c_j + t)\right) dt, \label{eq:gh27} \\
  \phi(x) &\le \omega x + \big(\phi(\xi) - \omega \xi\big)_+ \qquad \forall\,\omega,\xi \ge 0,\; x \in [0,\xi], \label{eq:gh28} \\
  \sum_{j=1}^m \int_{c_j}^\infty e^{-\lambda u}\bar{F}_j(u)\,du &\le \frac{1}{\lambda} L(\lambda B) - \sum_{j=1}^m \frac{r_j}{c_j}\int_0^{c_j} e^{-\lambda u}\,du \qquad \forall\,\lambda > 0. \label{eq:gh29}
\end{align}

\section{Time-Varying Jensen Alignment and Reduction to \texorpdfstring{$S = B = 1$}{S = B = 1}}\label{sec:ub-jensen}

Under~\eqref{eq:norm} and Lemma~\ref{lem:finite-caps}, define the discrete cap probability measure on $(0,\infty)$ by $\mu := \sum_{j=1}^m \frac{r_j}{S}\,\delta_{c_j}$ (kernel-proved to be a probability measure supported on $(0,\infty)$ in \texttt{capMeasure\_isProbabilityMeasure}, \texttt{capMeasure\_ae\_pos}), and set $U_1(t) := \int_{(0,\infty)} (c+t)^{-1}\,d\mu(c) > 0$, so $U(t) := \sum_{j=1}^m \frac{r_j}{c_j+t} = S\,U_1(t)$. Define $h(0) := 0$ and $h(x) := \phi(x)/x = 1 - (1-e^{-x})/x$ for $x > 0$. For any exponential mixture $w(u) = \sum_{k=1}^K a_k e^{-\lambda_k u}$ ($a_k \ge 0$, $\lambda_k > 0$, $\sum_k a_k = 1$) and measurable $\bar{U} : (0,\infty) \to [0,\infty)$ with $\bar{U}(t) \le 1/t$, define
\begin{equation}\label{eq:psi-def}
  \Psi_{w,\bar{U}}(c) := \int_0^\infty \frac{1}{c+t}\big(h(\bar{U}(t)) - w(c+t)\big)_+\,dt - \frac{1}{c}\int_0^c w(u)\,du \qquad (c \in [0,\infty)),
\end{equation}
with $\frac{1}{c}\int_0^c w(u)\,du|_{c=0} := 1$ and $\Psi_{w,\bar{U}}(\infty) := 0$ (continuous on $[0,\infty]$ by dominated convergence against $\min(\sum_k a_k\lambda_k, \frac{1}{2t^2}) \in L^1(0,\infty)$).

\begin{lemma}[Time-Varying Jensen Alignment and Monotonicity Reduction]\label{lem:jensen-mono}
Assume~\eqref{eq:norm} and~\eqref{eq:gh27}--\eqref{eq:gh29}. For every exponential mixture $w(u) = \sum_{k=1}^K a_k e^{-\lambda_k u}$ ($a_k \ge 0$, $\lambda_k > 0$, $\sum_k a_k = 1$) and every measurable $\bar{U}$ satisfying $U_1(t) \le \bar{U}(t) \le 1/t$ on $(0,\infty)$,
\begin{equation}\label{eq:master-reduced}
  \REV(\bm{X}) \le 1 + \sum_{k=1}^K \frac{a_k}{\lambda_k}\,L(\lambda_k) + \int_{(0,\infty)} \Psi_{w,\bar{U}}(c)\,d\mu(c).
\end{equation}
\end{lemma}

\begin{proof}
Fix $t > 0$ and write $q_j := \bar{F}_j(c_j+t) \le b_j := \frac{r_j}{c_j+t}$, $U = U(t) = \sum_j b_j$, and $w_j := w(c_j+t)$. Equivalently via pointwise Jensen weights $\pi_j(t) := b_j/U$ or directly via the chord bound for convex $\phi$ with $\phi(0) = 0$ on $[0, U]$:
\[
  \phi\!\left(\sum_{j=1}^m q_j\right)
  \le h(U)\sum_{j=1}^m q_j
  = \sum_{j=1}^m \Big(w_j q_j + (h(U) - w_j)q_j\Big)
  \le \sum_{j=1}^m \Big(w_j q_j + b_j\big(h(U) - w_j\big)_+\Big).
\]
Integrating over $t \in (0,\infty)$ (valid by Tonelli), substituting $u = c_j + t$, and applying~\eqref{eq:gh29} yields $\REV(\bm{X}) \le S + \sum_k \frac{a_k}{\lambda_k}L(\lambda_k B) + S\int \Psi_{w,U}(c)\,d\mu(c)$. Since $L$ is increasing and $B \le 1$, $L(\lambda_k B) \le L(\lambda_k)$; since $h$ is increasing ($h'(x) = (1-(1+x)e^{-x})/x^2 > 0$) and $U = SU_1 \le U_1 \le \bar{U}$, $\Psi_{w,U} \le \Psi_{w,\bar{U}}$; and since $w \le 1$, $1 + \Psi_{w,\bar{U}}(c) \ge 0$, so multiplying by $S \le 1$ gives~\eqref{eq:master-reduced}. (Kernel-proved from typed \texttt{GH26BaseInequalities} in \texttt{UpperBoundJensenAndDomain.lean}, with concrete $m = 1$ non-vacuity witness \texttt{gh26\_nonvacuous\_m1}.)
\end{proof}

\section{Stieltjes Reciprocal Concavity and Three-Moment Lens}\label{sec:ub-lens}

The reciprocal of a Stieltjes transform of a positive measure on $(0,\infty)$ is a classical complete Bernstein function and therefore concave and nondecreasing~\cite[Theorem~7.3]{SchillingSongVondracek12}. For completeness and formalization, we include the elementary Cauchy--Schwarz proof for any probability measure $\mu$ on $(0,\infty)$.

\begin{lemma}[Reciprocal Concavity of the Cap Stieltjes Transform]\label{lem:recip-conc}
Let $\mu$ be any probability measure on $(0,\infty)$, $U_1(t) := \int_{(0,\infty)} (c+t)^{-1}\,d\mu(c)$, and $F(t) := 1/U_1(t)$ for $t > 0$. Then $F$ is smooth on $(0,\infty)$ and satisfies: (a) $F(t) \ge t$; (b) $F'(t) \ge 1$ (so $F(0^+) := \lim_{t\downarrow 0} F(t) \ge 0$ exists); and (c) $F''(t) \le 0$, so $F$ is concave on $(0,\infty)$.
\end{lemma}

\begin{proof}
Since $c > 0$, $(c+t)^{-1} \le t^{-1}$, giving (a). For $t \ge \varepsilon > 0$, $(c+t)^{-k} \le \varepsilon^{-k}$ uniformly, so dominated convergence gives $U_1'(t) = -\int (c+t)^{-2}\,d\mu$ and $U_1''(t) = 2\int (c+t)^{-3}\,d\mu$. By Cauchy--Schwarz and $\int 1\,d\mu = 1$, $F'(t) = -U_1'(t)/U_1(t)^2 = (\int (c+t)^{-2}\,d\mu)(\int 1\,d\mu) / (\int (c+t)^{-1}\,d\mu)^2 \ge 1$, giving (b). Applying Cauchy--Schwarz to $(c+t)^{-1/2}$ and $(c+t)^{-3/2}$ gives $F''(t) = \frac{2}{U_1(t)^3}[(\int (c+t)^{-2}\,d\mu)^2 - (\int (c+t)^{-1}\,d\mu)(\int (c+t)^{-3}\,d\mu)] \le 0$, giving (c). (Kernel-proved in \texttt{UpperBoundRecipAndTree.lean}.)
\end{proof}

Fix $(l_1, l_2, l_3) := (\frac{3}{50}, \frac{7}{20}, \frac{9}{5})$, define $\gamma_i := U_1(l_i) \in (0, 1/l_i)$, and let $\mathcal{L} := \{\gamma(\mu)\}$ be the 3D moment lens with closure $\overline{\mathcal{L}} = \mathrm{conv}(\Gamma([0,\infty]))$ where $\Gamma(c) = ((c+l_1)^{-1},(c+l_2)^{-1},(c+l_3)^{-1})$ and $\Gamma(\infty) = (0,0,0)$.

\begin{corollary}[Harmonic-Secant Upper Envelope]\label{cor:ubar}
Let $\gamma = \gamma(\mu) \in \mathcal{L}$ and $0 < \gamma_i \le \bar{\gamma}_i \le 1/l_i$ ($i=1,2,3$) with $\bar{\gamma}_1 \ge \bar{\gamma}_2 \ge \bar{\gamma}_3$. Define $u_i(t) := \frac{l_i}{t}\bar{\gamma}_i$ ($0 < t \le l_i$), $u_i(t) := (1/\bar{\gamma}_i - l_i + t)^{-1}$ ($t \ge l_i$), and $s_i(t) := \bigl(\frac{l_{i+1}-t}{l_{i+1}-l_i}\frac{1}{\bar{\gamma}_i} + \frac{t-l_i}{l_{i+1}-l_i}\frac{1}{\bar{\gamma}_{i+1}}\bigr)^{-1}$ ($t \in [l_i, l_{i+1}]$). Setting $\bar{U}(t;\bar{\gamma}) := \min(\{1/t, u_1(t), u_2(t), u_3(t)\} \cup \{s_i(t) : t \in [l_i, l_{i+1}]\})$, we have $U_1(t) \le \bar{U}(t;\bar{\gamma}) \le 1/t$ for all $t > 0$, and $t \mapsto \bar{U}(t;\bar{\gamma})$ is continuous and nonincreasing on $(0,\infty)$.
\end{corollary}

\begin{proof}
Immediate from Lemma~\ref{lem:recip-conc}: concavity on $(0, l_i]$ with $F(0^+) \ge 0$ gives $F(t) \ge \frac{t}{l_i\bar{\gamma}_i}$; $F' \ge 1$ on $[l_i,\infty)$ gives $F(t) \ge 1/\bar{\gamma}_i - l_i + t$; concavity on $[l_i,l_{i+1}]$ gives $F(t) \ge 1/s_i(t)$; continuity at $l_i, l_{i+1}$ holds since $s_i(l_i) = \bar{\gamma}_i \ge u_0(l_i)$ and $s_i(l_{i+1}) = \bar{\gamma}_{i+1} \ge u_0(l_{i+1})$. (Kernel-proved in \texttt{U1\_le\_Ubar\_le\_inv}.)
\end{proof}

\begin{lemma}[Valid Inequalities on the 3D Moment Lens]\label{lem:lens}
Every $\gamma = (\gamma_1,\gamma_2,\gamma_3) \in \mathcal{L}$ satisfies:
\begin{enumerate}
  \item[(i)] $\gamma_2 \ge \frac{l_1}{l_2}\gamma_1$ and $\gamma_3 \ge \frac{l_2}{l_3}\gamma_2$;
  \item[(ii)] $\gamma_j \le \varphi_{ij}(\gamma_i) := \frac{\gamma_i}{1 + (l_j - l_i)\gamma_i}$ for $1 \le i < j \le 3$, and for every $g_0 \ge 0$, $\gamma_j \le \frac{g_0}{1+(l_j-l_i)g_0} + \frac{\gamma_i - g_0}{(1+(l_j-l_i)g_0)^2}$;
  \item[(iii)] $\gamma_2 \ge A_{\mathrm{sec}}\gamma_1 + B_{\mathrm{sec}}\gamma_3$, where $A_{\mathrm{sec}} = \frac{l_1(l_3-l_2)}{l_2(l_3-l_1)}$, $B_{\mathrm{sec}} = \frac{l_3(l_2-l_1)}{l_2(l_3-l_1)}$;
  \item[(iv)] $\gamma_2 \le (1-\theta_{13})\gamma_1 + \theta_{13}\gamma_3$, where $\theta_{13} = \frac{l_2-l_1}{l_3-l_1}$;
  \item[(v)] $\gamma_2 \le ((1-\theta_{13})/\gamma_1 + \theta_{13}/\gamma_3)^{-1}$, hence for every $p, q > 0$, $\gamma_2 \le \frac{(1-\theta_{13})p^2}{((1-\theta_{13})p + \theta_{13}q)^2}\gamma_1 + \frac{\theta_{13}q^2}{((1-\theta_{13})p + \theta_{13}q)^2}\gamma_3$.
\end{enumerate}
\end{lemma}

\begin{proof}
(i) $(c+l_1)/(c+l_2) \ge l_1/l_2$; (ii) $F(l_j) \ge F(l_i) + (l_j-l_i)$ and concavity of $\varphi_{ij}$; (iii) concavity of $t \mapsto t/(c+t)$; (iv) convexity of $t \mapsto (c+t)^{-1}$; (v) concavity of $F$ plus Cauchy--Schwarz. (All five and both tangents are kernel-proved in \texttt{UpperBoundRecipAndTree.lean}.)
\end{proof}

\section{Farkas Decoupling, Quadrature Tails, and Proof of Theorem~\texorpdfstring{\ref{thm:ub-main}}{1.1}}\label{sec:ub-farkas}

With $(l_1,l_2,l_3) = (\frac{3}{50},\frac{7}{20},\frac{9}{5})$, $\mathcal{L} \subset R_0 := [0,17] \times [0,3] \times [0,\frac{5}{8}]$. On any box $R = \prod_{i=1}^3 [\underline{g}_i,\overline{g}_i] \subset R_0$ with midpoint $g_{i,m} > 0$, Lemma~\ref{lem:lens} gives $14$ linear inequalities $D\gamma \le d$ ($6$ box, $4$ global, $3$ concave tangents at $g_{i,m}$, $1$ harmonic tangent at $p = \max(g_{3,m}, 10^{-6}), q = \max(g_{1,m}, 10^{-6})$). Deterministic upper-corner tightening initializes $\bar{\gamma}_i^{(0)} := \min(\overline{g}_i, 1/l_i)$, iterates $\bar{\gamma}_2 \leftarrow \min(\bar{\gamma}_2, \varphi_{12}(\bar{\gamma}_1))$, $\bar{\gamma}_3 \leftarrow \min(\bar{\gamma}_3, \varphi_{23}(\bar{\gamma}_2), \varphi_{13}(\bar{\gamma}_1))$, $\bar{\gamma}_2 \leftarrow \min(\bar{\gamma}_2, ((1-\theta_{13})/\bar{\gamma}_1 + \theta_{13}/\bar{\gamma}_3)^{-1})$ three times plus a final monotone pass, guaranteeing $\gamma_i \le \bar{\gamma}_i \le 1/l_i$ and $\bar{\gamma}_1 \ge \bar{\gamma}_2 \ge \bar{\gamma}_3 > 0$.

\begin{lemma}[Box-Farkas Certificate Reduction]\label{lem:farkas}
For any normalized instance with $\gamma(\mu) \in R \cap \mathcal{L}$, tightened corner $\bar{\gamma}$, exponential mixture $w$, $\eta_0 \in \R$, and $y \in \R_{\ge 0}^{14}$ ($\eta := D^\top y$),
\begin{equation}\label{eq:box-cert}
  \REV(\bm{X}) \le 1 + \sum_{k=1}^K \frac{a_k}{\lambda_k}\,L(\lambda_k) + \eta_0 + d^\top y
                   + \sup_{c \in [0,\infty]} \left[\Psi_{w,\bar{U}(\cdot;\bar{\gamma})}(c) - \eta_0 - \sum_{i=1}^3 \frac{\eta_i}{c + l_i}\right]_+.
\end{equation}
\end{lemma}

\begin{proof}
Since $D\gamma(\mu) \le d$ and $y \ge 0$, $\eta^\top\gamma(\mu) \le d^\top y$; integrating the pointwise bound on $\Psi_{w,\bar{U}}(c)$ against $\mu$ and applying Lemma~\ref{lem:jensen-mono} yields~\eqref{eq:box-cert}. (Kernel-proved in \texttt{polytope\_valid}, \texttt{farkas\_decoupling}.)
\end{proof}

\begin{lemma}[Analytic Tail and Monotonicity Bounds]\label{lem:tails}
For any exponential mixture $w(u) = \sum_{k=1}^K a_k e^{-\lambda_k u}$ and nonincreasing $\bar{U}(t) \le 1/t$:
(1)~for any grid $0 = s_0 < \dots < s_N$, convexity of $f_\lambda(s) := e^{-\lambda s}(1+s)^{-2}$ gives $L(\lambda) \le \lambda - \ln(\sum_{r=0}^{N-1}(s_{r+1}-s_r)f_\lambda(\frac{s_r+s_{r+1}}{2}))$;
(2)~$\int_0^{t_0} (c+t)^{-1}(h(\bar{U}(t))-w(c+t))_+\,dt \le t_0 \sum_k a_k\lambda_k$;
(3)~$\int_T^\infty (c+t)^{-1}(h(\bar{U}(t))-w(c+t))_+\,dt \le \frac{1}{2T}$ (using $h(x) \le x/2$);
(4)~on $[c_a,c_b] \times [t_j,t_{j+1}]$, the Hermite--Hadamard trapezoid bound for convex $t \mapsto (c_a+t)^{-1}$ yields $\int_{t_j}^{t_{j+1}} \frac{(h(\bar{U}(t))-w(c+t))_+}{c+t}\,dt \le (h(\bar{U}(t_j))-w(c_b+t_{j+1}))_+\frac{t_{j+1}-t_j}{2}((c_a+t_j)^{-1}+(c_a+t_{j+1})^{-1})$;
(5)~for all $c \ge C_{\max} > 0$, $\Psi_{w,\bar{U}}(c) \le \ln(1+\frac{1}{2C_{\max}}) + \frac{\ln(1+2C_{\max})}{2C_{\max}}$ and $\frac{\eta_i}{c+l_i} \ge \min(0, \frac{\eta_i}{C_{\max}+l_i})$.
\end{lemma}

\section{Concentrating Bundle \texorpdfstring{$B$}{B}, Continuum \texorpdfstring{$\BREV$}{BREV} Split, and Revenue Bounds}\label{sec:lb-bundle}

Following~\cite[Appendix~D]{GiannakopoulosHahn26}, for any integer $N_b \ge 2$ let $B$ contain $N_b^2$ i.i.d.\ items (independent of $A$) supported on $\{k\Delta : k \in [N_b]\}$ with $\Delta := \frac{S_A}{N_b^2 H_{N_b}}$, $H_{N_b} := \sum_{k=1}^{N_b} \frac{1}{k}$, and $\Pr[Y \ge k\Delta] = \frac{1}{k}$. Write $Z_A := \sum_{j\in A} X_j$ and $Z_B := \sum_{j\in B} Y_j$.

\begin{lemma}[Exact Moments of $B$ and $\SREV/\BREV$ Comparison]\label{lem:b-bundle}
For every integer $N_b \ge 2$:
(1)~$\REV(Y) = \Delta$, $\SREV(B) = S_A/H_{N_b}$, and $\SREV(A\cup B) = S_A + S_A/H_{N_b}$;
(2)~$\E[Z_B] = S_A$ and $\Var(Z_B) = \frac{S_A^2}{N_b^2 H_{N_b}^2}(2N_b - H_{N_b} - H_{N_b}^2) < \frac{2S_A^2}{N_b H_{N_b}^2}$;
(3)~if $\sup_{p \ge 0} p\Pr[S_A + Z_A \ge p] = S_A$, then $\BREV(A\cup B) \le \SREV(A\cup B)$.
\end{lemma}

\begin{proof}
Items~(1)--(2) follow by summation of survival probabilities; for~(3), fixing $x \ge 0$ and $p > 0$: if $x \le S_A$ then $p\Pr[x+Z_A \ge p] \le S_A$; if $x > S_A$ and $p > x$, setting $p' = p - (x-S_A) > S_A$ gives $p\Pr[S_A + Z_A \ge p'] \le (p/p')S_A = S_A + \frac{S_A}{p'}(x-S_A) < x$. Conditioning on $Z_B$ and using Cauchy--Schwarz plus $\sqrt{2/N_b} \le 1$ gives $\BREV(A\cup B) \le \E[\max\{S_A, Z_B\}] \le S_A + \sqrt{\Var(Z_B)} < S_A + \frac{S_A}{H_{N_b}}\sqrt{2/N_b} \le \SREV(A\cup B)$. (Kernel-proved in \texttt{LowerBoundChebyshevAndWitness.lean}.)
\end{proof}

\begin{lemma}[Finite-Grid and Chernoff Continuum Bounds on {$\sup_{p \ge 0} p\Pr[S_A + Z_A \ge p]$}]\label{lem:brev-split}
Let $P_0 := 55$ and $K_{\mathrm{cut}} := \lceil D(P_0 - S_A)\rceil = 5013$ ($S_A + K_{\mathrm{cut}}/D = 55.0025 \ge P_0$). Then:
\begin{enumerate}
  \item $\sup_{p \in (S_A, P_0]} p\Pr[S_A + Z_A \ge p] \le \max_{1 \le k \le K_{\mathrm{cut}}} (S_A + \frac{k}{D})\Pr[Z_A \ge \frac{k}{D}]$;
  \item for any $t > 0$ with $P_0 \ge 1/t$ (we take $t = \frac{1}{10}$), $\sup_{p \ge P_0} p\Pr[S_A + Z_A \ge p] \le P_0 e^{-t(P_0-S_A)} \prod_{g=1}^8 (\E[e^{tX_g}])^{N_a}$.
\end{enumerate}
\end{lemma}

\begin{proof}
(1)~$Z_A \in \frac{1}{D}\mathbb{Z}_{\ge 0}$ so $\{Z_A \ge p - S_A\} = \{Z_A \ge \lceil D(p-S_A)\rceil/D\}$. (2)~Markov's inequality on $e^{tZ_A}$ and nonincreasingness of $pe^{-tp}$ for $p \ge 1/t$.
\end{proof}

\paragraph{Mechanism $\mathcal{M}_0$, tie-breaking invariance, and revenue bounds.}
Offer bundle $B$ at price $p_B := S_A - S_A/H_{N_b} > 0$, and independently offer every subset $T \subseteq A$ at price $0$ ($T = \emptyset$) or $F + \sum_{j\in T} p_j$ ($T \neq \emptyset$). Letting $E_A$ and $R_B(N_b)$ be the seller-favoring expected revenues on $A$ and $B$:

\begin{lemma}[Tie-Breaking Invariance via $\varepsilon$-Perturbation]\label{lem:tiebreak}
For every $\delta > 0$, lowering $p_B \mapsto p_B - \varepsilon_1$, $p_j \mapsto p_j - \varepsilon_2$, and $F \mapsto F - \varepsilon_3$ (with $\varepsilon_1 \in (0, p_B)$, $\varepsilon_2 \in (0, 0.07)$, $\varepsilon_3 \in (0, 7)$, $\varepsilon_1 + 8N_a\varepsilon_2 + \varepsilon_3 \le \delta$) produces a deterministic DSIC/IR mechanism $\mathcal{M}_\delta$ whose expected revenue under \emph{every} buyer tie-breaking rule is at least $E_A + R_B(N_b) - \delta$.
\end{lemma}

\begin{proof}
Because $V_g \cap (0, p_g) = \emptyset$, zero-valued items have strictly negative surplus $-(p_g - \varepsilon_2) < 0$ while every item with $X_j \ge p_j$ has surplus $\ge \varepsilon_2 > 0$; thus $Z_B \ge p_B$ and $U_A \ge F$ become strict utility gains under $\mathcal{M}_\delta$, losing at most $\varepsilon_1 + 8N_a\varepsilon_2 + \varepsilon_3 \le \delta$. (Kernel-proved in \texttt{LowerBoundDSIC.lean}.)
\end{proof}

Chebyshev's inequality gives $\Pr[Z_B < p_B] < 2/N_b$, hence $R_B(N_b) \ge (S_A - S_A/H_{N_b})(1 - 2/N_b)$. On $A$, all-paid pricing gives unconditional payment $\sum_{g=1}^8 N_a p_g \frac{D r_g}{N_a k_{\min,g}} = S_A$, so
\[

\]
